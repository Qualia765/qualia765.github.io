#define DEFINE_VECTOR(type, name) \
     struct name { \
           type *data; \
           size_t size; \
           size_t capacity; \
     }; \
     static inline void name##_initialize(struct name *vec) { \
           vec->size = 0; \
           vec->capacity = 8; \
           vec->data = malloc(8 * sizeof(type)); \
     } \
     static inline void name##_set_capacity(struct name *vec, size_t new_capacity) { \
           assert(vec->data); \
           if (vec->size > new_capacity) vec->size = new_capacity; \
           vec->capacity = new_capacity; \
           vec->data = realloc(vec->data, new_capacity * sizeof(type)); \
     } \
     static inline void name##_free(struct name *vec) { \
           assert(vec->data); \
           free(vec->data); \
           vec->data = NULL; \
           vec->capacity = 0; \
           vec->size = 0; \
     } \
     static inline size_t name##_enbiggen(struct name *vec) { \
           assert(vec->data); \
           if (vec->size == vec->capacity) { \
                 vec->capacity = (vec->capacity << 1) + (vec->capacity >> 1); \
                 vec->data = realloc(vec->data, vec->capacity * sizeof(type)); \
           } \
           return vec->size++; \
     } \
     static inline void name##_push(struct name *vec, type value) { \
           assert(vec->data); \
           size_t last = name##_enbiggen(vec); \
           vec->data[last] = value; \
     } \
     static inline type * name##_top(struct name *vec) { \
           assert(vec->data); \
           assert(vec->size > 0); \
           return &vec->data[vec->size-1]; \
     }

void set_article_name(char * new_article_name) {
     size_t new_article_name_len = strlen(new_article_name);
     if (new_article_name_len > article_name_capacity) {
           article_name = realloc(article_name, (new_article_name_len+1)*sizeof(char));
           article_name_capacity = new_article_name_len;
     }
     memcpy(article_name, new_article_name, new_article_name_len+1);
     article_name_len = new_article_name_len;
}

char * my_strdup(char *src) {
     size_t mem_size = (strlen(src)+1)*sizeof(char);
     char *result = malloc(mem_size);
     memcpy(result, src, mem_size);
     return result;
}

     section->section_name = section_name;

     section->min_words = INT_MAX;
     section->max_words = 0;
     section->mean_words = 0;
     section->std_words = 0;
     section->awards = 0;
     struct int_vector *vec = &(section->words_per_sentence);

}

     struct section_data *new_section = &(all_section_data.data[last]);
     section_data_initialize(new_section, section_name);
}

// ~~~~~~~~~ COLLECT SUMMARY STUFF ~~~~~~~~~

void analyze_section(struct section_data *section) {
     size_t n = section->words_per_sentence.size;
     float n_float = (float)n;
     for (size_t i=0; i<n; ++i) {
           int words = section->words_per_sentence.data[i];
           section->total_words += words;
           if (section->min_words > words) section->min_words = words;
           if (section->max_words < words) section->max_words = words;
     }
     section->mean_words = ((float)section->total_words) / n_float;
     float variance = 0;
     for (size_t i=0; i<n; ++i) {
           float words = (float)(section->words_per_sentence.data[i]);
           float difference = words - section->mean_words;
           variance += difference * difference;
     }
     section->std_words = sqrt(variance / n_float);
}

char* append_string_with_newline_in_middle(char *bottom, const char *top) {
     size_t old_bottom_len = strlen(bottom);
     size_t top_len = strlen(top);
     if (old_bottom_len==0) {
           bottom = realloc(bottom, (top_len+1)*sizeof(char));
           memcpy(bottom, top, top_len+1);
     } else {
           size_t new_len = old_bottom_len + 1 + top_len; // +1 for \n
           bottom = realloc(bottom, (new_len+1)*sizeof(char));
           bottom[old_bottom_len] = '\n';
           // name+old_bottom_len+1 to get pass newline
           // top_len+1 to copy its \0
           memcpy(bottom+old_bottom_len+1, top, top_len+1);

     }
     return bottom;
}

void collect_summary(bool is_article) {
     if (is_article) set_article_name(yytext);
     struct section_data *current_section = section_data_vector_top(&all_section_data);
     if (current_section->words_per_sentence.size != 0) {
           analyze_section(current_section);
           char * section_name = my_strdup(article_name);
           if (!is_article) section_name = append_string_with_newline_in_middle(section_name, yytext);
           all_section_data_enbiggen_and_initialize(section_name);

     } else {
           if (!is_article) {
                 // When the section is after the section with not sentences in between,
                 // then append the now seen section name to section_name
                 current_section->section_name =
                       append_string_with_newline_in_middle(current_section->section_name, yytext);
           }
     }
     current_words = 0;
}

static int section_metric_total(const struct section_data *section) { return section->total_words; }
static int section_metric_sentences(const struct section_data *section) { return section->words_per_sentence.size; }
static float section_metric_mean(const struct section_data *section) { return section->mean_words; }
static int section_metric_min(const struct section_data *section) { return section->min_words; }
static int section_metric_max(const struct section_data *section) { return section->max_words; }
static int section_metric_range(const struct section_data *section) { return section->max_words - section->min_words; }
static float section_metric_std(const struct section_data *section) { return section->std_words; }

typedef int (*section_metric_int_fn)(const struct section_data *);

static void assign_awards_single_int(section_metric_int_fn get, uint16_t min_award, uint16_t max_award) {
     size_t min_index = 0;
     size_t max_index = 0;
     int min_val = get(&(all_section_data.data[0]));
     int max_val = min_val;
     for (size_t i = 1; i < all_section_data.size; ++i) {
           int val = get(&(all_section_data.data[i]));
           if (val < min_val) {
                 min_val = val;
                 min_index = i;
           }
           if (val > max_val) {
                 max_val = val;
                 max_index = i;
           }
     }
     all_section_data.data[min_index].awards |= min_award;
     all_section_data.data[max_index].awards |= max_award;
}

typedef float (*section_metric_float_fn)(const struct section_data *);

static void assign_awards_single_float(
     section_metric_float_fn get, const uint16_t min_award, const uint16_t max_award
) {
     size_t min_index = 0;
     size_t max_index = 0;
     float min_val = get(&(all_section_data.data[0]));
     float max_val = min_val;
     for (size_t i = 1; i < all_section_data.size; ++i) {
           float val = get(&(all_section_data.data[i]));
           if (val < min_val) {
                 min_val = val;
                 min_index = i;
           }
           if (val > max_val) {
                 max_val = val;
                 max_index = i;
           }
     }
     all_section_data.data[min_index].awards |= min_award;
     all_section_data.data[max_index].awards |= max_award;
}

void assign_awards(void) {
     if (all_section_data.size == 0) return;
     assign_awards_single_int(section_metric_total,        AWARD_MIN_TOTAL,        AWARD_MAX_TOTAL);
     assign_awards_single_int(section_metric_sentences,    AWARD_MIN_SENTENCES,    AWARD_MAX_SENTENCES);
     assign_awards_single_float(section_metric_mean,       AWARD_MIN_MEAN,         AWARD_MAX_MEAN);
     assign_awards_single_int(section_metric_min,          AWARD_MIN_MIN,          AWARD_MAX_MIN);
     assign_awards_single_int(section_metric_max,          AWARD_MIN_MAX,          AWARD_MAX_MAX);
     assign_awards_single_int(section_metric_range,        AWARD_MIN_RANGE,        AWARD_MAX_RANGE);
     assign_awards_single_float(section_metric_std,        AWARD_MIN_STD,          AWARD_MAX_STD);
}

//     ~~~ announce awards stuff ~~~

void print_award(const uint16_t award, const struct section_data *section) {
     switch (award) {
           case AWARD_MIN_TOTAL:
                 printf("least words (%d words)", section->total_words);
           break; case AWARD_MAX_TOTAL:
                 printf("most words (%d words)", section->total_words);
           break; case AWARD_MIN_SENTENCES:
                 printf("fewest sentences (%d sentences)", (int)(section->words_per_sentence.size));
           break; case AWARD_MAX_SENTENCES:
                 printf("most sentences (%d sentences)", (int)(section->words_per_sentence.size));
           break; case AWARD_MIN_MEAN:
                 printf("shortest sentences on average (%.2f words per sentence)", section->mean_words);
           break; case AWARD_MAX_MEAN:
                 printf("longest sentences on average (%.2f words per sentence)", section->mean_words);
           break; case AWARD_MIN_MIN:
                 printf("shortest sentence (%d words)", section->min_words);
           break; case AWARD_MAX_MIN:
                 printf("longest shortest sentence (%d words)", section->min_words);
           break; case AWARD_MIN_MAX:
                 printf("shortest longest sentence (%d words)", section->max_words);
           break; case AWARD_MAX_MAX:
                 printf("longest sentence (%d words)", section->max_words);
           break; case AWARD_MIN_RANGE:
                 printf("shortest range of length of sentences (%d word range)", section->max_words-section->min_words);
           break; case AWARD_MAX_RANGE:
                 printf("longest range of length of sentences (%d word range)", section->max_words-section->min_words);
           break; case AWARD_MIN_STD:
                 printf("smallest standard deviation of length of sentences (%.2f words std)", section->std_words);
           break; case AWARD_MAX_STD:
                 printf("largest standard deviation of length of sentences (%.2f words std)", section->std_words);
           break; default: printf("unknown interesting metric");
           break;
     }
}

void announce_awards(void) {
     for (size_t i=0; i<all_section_data.size; ++i) {
           struct section_data *current = &(all_section_data.data[i]);
           if (current->awards == 0) continue;
           printf("============\n%s\n\n", current->section_name);
           bool already_printed_award = false;
           for (size_t j=0; j<AWARD_COUNT; ++j) {
                 if ((current->awards) & (1<<j)) {
                       if (already_printed_award) printf("\n& ");
                       already_printed_award = true;
                       print_award(1 << j, current);
                 }
           }
           printf("\n\n");
     }
}

//     ~~~ final summary stuff ~~~

void announce_over_all_statistics(void) {
     printf("============\nOverall Statistics\n\n");

     int total_words = 0, total_sentences = 0;

     for (size_t i=0; i<all_section_data.size; ++i) {
           struct section_data *current = &(all_section_data.data[i]);
           total_words += current->total_words;
           total_sentences += current->words_per_sentence.size;
     }

     printf("Words: %d\nSentences: %d\n", total_words, total_sentences);

     double mean = ((double)total_words) / ((double)total_sentences);
     printf("Mean: %.2lf\n", mean);

     double variance = 0;
     for (size_t i=0; i<all_section_data.size; ++i) {
           struct section_data *current = &(all_section_data.data[i]);
           for (size_t j=0; j<current->words_per_sentence.size; ++j) {
                 double words = (double)(current->words_per_sentence.data[j]);
                 double difference = words - mean;
                 variance += difference * difference;
           }
     }

     double std = sqrt(variance / ((double)total_sentences));

     printf("Standard Deviation: %.2lf\n\n", std);
}
